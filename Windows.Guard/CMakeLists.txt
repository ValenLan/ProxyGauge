cmake_minimum_required(VERSION 3.24)

project(ProxyGaugeGuard LANGUAGES CXX)

add_executable(ProxyGauge.Guard Guard.cpp)
target_compile_features(ProxyGauge.Guard PRIVATE cxx_std_20)
target_compile_definitions(ProxyGauge.Guard PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ProxyGauge.Guard PRIVATE /W4 /WX /permissive- /utf-8)
target_link_libraries(ProxyGauge.Guard PRIVATE advapi32 fwpuclnt iphlpapi ws2_32 windowsapp)

install(TARGETS ProxyGauge.Guard RUNTIME DESTINATION .)

include(CTest)
if(BUILD_TESTING)
    add_executable(ProxyGauge.Guard.SelectionTests GuardSelectionTests.cpp)
    target_compile_features(ProxyGauge.Guard.SelectionTests PRIVATE cxx_std_20)
    target_compile_options(ProxyGauge.Guard.SelectionTests PRIVATE /W4 /WX /permissive- /utf-8)
    add_test(NAME automatic-core-handover COMMAND ProxyGauge.Guard.SelectionTests)
    add_executable(ProxyGauge.Guard.TunTests GuardTunControllerTests.cpp)
    target_compile_features(ProxyGauge.Guard.TunTests PRIVATE cxx_std_20)
    target_compile_definitions(ProxyGauge.Guard.TunTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(ProxyGauge.Guard.TunTests PRIVATE /W4 /WX /permissive- /utf-8)
    target_link_libraries(ProxyGauge.Guard.TunTests PRIVATE windowsapp)
    add_test(NAME tun-controller-evidence COMMAND ProxyGauge.Guard.TunTests)
    set_tests_properties(tun-controller-evidence PROPERTIES TIMEOUT 10)
endif()
